import base64 import numpy as np import io from PIL import Image from flask import request, Flask, render_template import tensorflow as tf from keras.preprocessing.image import img_to_array # Additional imports import h5py import json # Create a Flask application instance app = Flask(__name__) # Function to fix layer 0 configuration in the model file def fix_layer0(filename, batch_input_shape, dtype): with h5py.File(filename, 'r+') as f: model_config = json.loads(f.attrs['model_config']) layer0 = model_config['config']['layers'][0]['config'] layer0['batch_input_shape'] = batch_input_shape layer0['dtype'] = dtype f.attrs['model_config'] = json.dumps(model_config).encode('utf-8') # Function to load the Keras model def get_model(): global model, graph model = tf.keras.models.load_model('cnn_model.h5') print(" * Model Loaded!!") graph = tf.get_default_graph() # Preprocess the image before passing it to the model def preprocess_image(image, target_size): if image.mode != "RGB": image = image.convert("RGB") image = image.resize(target_size) image = img_to_array(image) image = np.expand_dims(image, axis=0) return image # Fix layer 0 and load the model print(" * Loading keras model.....") get_model() # Define the route to render the upload form @app.route("/") def upload_form(): return render_template("upload.html", prediction=None) # Define a route for prediction @app.route("/predict", methods=["POST"]) def predict(): # Get the uploaded image file file = request.files['image'] # Preprocess the image and make predictions with graph.as_default(): image = Image.open(file) processed_image = preprocess_image(image, target_size=(224, 224)) prediction = model.predict(processed_image).tolist() # Convert prediction to human-readable labels labels = ["No Diabetic Retinopathy", "Mild Diabetic Retinopathy", "Moderate Diabetic Retinopathy", "Severe Diabetic Retinopathy", "Proliferative Diabetic Retinopathy"] prediction_label = labels[np.argmax(prediction)] # Prepare and return the response JSON response = { 'prediction': prediction_label } return render_template("upload.html", prediction=prediction_label) # Run the Flask application if __name__ == "__main__": app.run(debug=True)
